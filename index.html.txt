<script>
  let mode = 'exkl', rate = 25;

  // Show iOS install hint only in Safari browser (not already installed)
  const isIOS = /iphone|ipad|ipod/i.test(navigator.userAgent);
  const isStandalone = window.navigator.standalone === true;
  const isSafari = /^((?!chrome|android).)*safari/i.test(navigator.userAgent);
  if (isIOS && !isStandalone && isSafari) {
    document.getElementById('install-banner').style.display = 'flex';
  }

  function setMode(m) {
    mode = m;
    document.getElementById('btn-exkl').classList.toggle('active', m === 'exkl');
    document.getElementById('btn-inkl').classList.toggle('active', m === 'inkl');
    document.getElementById('input-label').textContent =
      m === 'exkl' ? 'Belopp exkl. moms' : 'Belopp inkl. moms';
    calculate();
  }

  function setRate(r) {
    rate = r;
    [25, 12, 6].forEach(v =>
      document.getElementById('rate-' + v).classList.toggle('active', v === r));
    document.getElementById('moms-label').textContent = 'Moms (' + r + '%)';
    calculate();
  }

  function fmt(n) {
    return isNaN(n) ? '—' :
      n.toLocaleString('sv-SE', { minimumFractionDigits: 2, maximumFractionDigits: 2 }) + ' kr';
  }

  function pop(id) {
    const el = document.getElementById(id);
    el.classList.remove('pop');
    void el.offsetWidth;
    el.classList.add('pop');
  }

  function calculate() {
    const raw = parseFloat(document.getElementById('amount').value);
    if (!raw || raw <= 0) {
      ['moms-value','base-value','total-value'].forEach(id =>
        document.getElementById(id).textContent = '—');
      return;
    }
    let base, moms, total;
    if (mode === 'exkl') {
      base = raw; moms = raw * (rate/100); total = raw + moms;
    } else {
      total = raw; base = raw / (1 + rate/100); moms = total - base;
    }
    document.getElementById('moms-value').textContent  = fmt(moms);
    document.getElementById('base-value').textContent  = fmt(base);
    document.getElementById('total-value').textContent = fmt(total);
    pop('moms-value'); pop('total-value');
  }

  // Register inline service worker for offline support
  if ('serviceWorker' in navigator) {
    const sw = `
      const CACHE = 'momskalkyl-v1';
      self.addEventListener('install', e => {
        e.waitUntil(caches.open(CACHE).then(c => c.addAll(['.'])));
        self.skipWaiting();
      });
      self.addEventListener('activate', e => {
        e.waitUntil(caches.keys().then(keys =>
          Promise.all(keys.filter(k => k !== CACHE).map(k => caches.delete(k)))));
        self.clients.claim();
      });
      self.addEventListener('fetch', e => {
        e.respondWith(caches.match(e.request).then(r => r || fetch(e.request)));
      });
    `;
    const blob = new Blob([sw], {type: 'application/javascript'});
    navigator.serviceWorker.register(URL.createObjectURL(blob));
  }
</script>

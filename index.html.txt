<script>
  let mode = 'exkl';
  let rate = 25;

  function setMode(m) {
    mode = m;
    document.getElementById('btn-exkl').classList.toggle('active', m === 'exkl');
    document.getElementById('btn-inkl').classList.toggle('active', m === 'inkl');

    if (m === 'exkl') {
      document.getElementById('input-label').textContent = 'Belopp exkl. moms';
      document.getElementById('base-label').textContent = 'Exkl. moms';
      document.getElementById('total-label').textContent = 'Totalt inkl. moms';
    } else {
      document.getElementById('input-label').textContent = 'Belopp inkl. moms';
      document.getElementById('base-label').textContent = 'Exkl. moms';
      document.getElementById('total-label').textContent = 'Totalt inkl. moms';
    }
    calculate();
  }

  function setRate(r) {
    rate = r;
    [25, 12, 6].forEach(v => {
      document.getElementById('rate-' + v).classList.toggle('active', v === r);
    });
    document.getElementById('moms-label').textContent = 'Moms (' + r + '%)';
    calculate();
  }

  function fmt(n) {
    if (isNaN(n)) return '—';
    return n.toLocaleString('sv-SE', { minimumFractionDigits: 2, maximumFractionDigits: 2 }) + ' kr';
  }

  function animPop(id) {
    const el = document.getElementById(id);
    el.classList.remove('pop');
    void el.offsetWidth;
    el.classList.add('pop');
  }

  function calculate() {
    const raw = parseFloat(document.getElementById('amount').value);

    if (!raw || raw <= 0) {
      document.getElementById('moms-value').textContent = '—';
      document.getElementById('base-value').textContent = '—';
      document.getElementById('total-value').textContent = '—';
      return;
    }

    let base, moms, total;

    if (mode === 'exkl') {
      base  = raw;
      moms  = raw * (rate / 100);
      total = raw + moms;
    } else {
      total = raw;
      base  = raw / (1 + rate / 100);
      moms  = total - base;
    }

    document.getElementById('moms-value').textContent  = fmt(moms);
    document.getElementById('base-value').textContent  = fmt(base);
    document.getElementById('total-value').textContent = fmt(total);

    animPop('moms-value');
    animPop('total-value');
  }
</script>
